const lastF  = bid => [...DB.fali].filter(f=>f.bufId===bid).sort((a,b)=>b.date.localeCompare(a.date))[0];
const is3m   = bid => { const f=lastF(bid); return f?dAgo(f.date)>90:false; };
const isK    = bid => lastF(bid)?.status==='K';
const isP    = bid => lastF(bid)?.status==='P';

function sBadge(s){
  if(s==='R') return '<span class="badge b-red">↩ R</span>';
  if(s==='P') return '<span class="badge b-blue">✅ P</span>';
  if(s==='K') return '<span class="badge b-amber">🔵 Khali</span>';
  if(s==='Bimar') return '<span class="badge b-amber">🤒 Bimar</span>';
  return '<span class="badge b-gray">—</span>';
}
function pBadge(paid,total){
  if(!total) return '';
  if(paid>=total) return '<span class="badge b-green">PAID</span>';
  if(paid>0)      return '<span class="badge b-amber">PART</span>';
  return '<span class="badge b-red">DUE</span>';
}

function nav(page){
  curPage=page;
  // Desktop nav items
  document.querySelectorAll('.nav-item').forEach(b=>b.classList.remove('active'));
  const navBtn=[...document.querySelectorAll('.nav-item')].find(b=>b.getAttribute('onclick')?.includes(`'${page}'`));
  if(navBtn) navBtn.classList.add('active');
  // Mobile tabs
  document.querySelectorAll('.mt-item').forEach(b=>b.classList.remove('active'));
  const mBtn=[...document.querySelectorAll('.mt-item')].find(b=>b.getAttribute('onclick')?.includes(`'${page}'`));
  if(mBtn) mBtn.classList.add('active');
  // Pages
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
  const pageEl=document.getElementById('page-'+page);
  if(pageEl){ pageEl.classList.add('active'); renderPage(page); }
  document.getElementById('page-title').textContent=PAGE_TITLES[page]||page;
  document.getElementById('content').scrollTop=0;
  updateAlertBadge();
}

}

/* ═══════════════════════════════════════════════════════
   P&L CALCULATOR
═══════════════════════════════════════════════════════ */
function calcPL(m){
  const mk=(DB.milkCust||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.value,0);
  const sl=(DB.slaughter||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.amount,0);
  const ex=(DB.expenses||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.amount,0);
  const cb=(DB.careBills||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.bill,0);
  const tr=(DB.transport||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.total,0);
  const lb=(DB.salaries||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.sal,0);
  const pur=(DB.purchases||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.amount,0);
  const dl=(DB.deaths||[]).filter(r=>r.date.startsWith(m)).reduce((s,r)=>s+r.val,0);
  const inc=mk+sl, exp=ex+cb+tr+lb+pur, net=inc-exp-dl;
  return `<div class="pl-block">
    <div class="pl-row"><span class="pl-label">🥛 Milk Income</span><span class="pl-value tg">₹${fmt(mk)}</span></div>
    <div class="pl-row"><span class="pl-label">🔪 Slaughter</span><span class="pl-value tg">₹${fmt(sl)}</span></div>
    <div class="pl-row"><span class="pl-label">🌾 Feed Stock</span><span class="pl-value tr">₹${fmt(0)}</span></div>
    <div class="pl-row"><span class="pl-label">👤 Caretakers</span><span class="pl-value tr">₹${fmt(cb)}</span></div>
    <div class="pl-row"><span class="pl-label">🚛 Transport</span><span class="pl-value tr">₹${fmt(tr)}</span></div>
    <div class="pl-row"><span class="pl-label">👷 Labour</span><span class="pl-value tr">₹${fmt(lb)}</span></div>
    <div class="pl-row"><span class="pl-label">💡 Expenses</span><span class="pl-value tr">₹${fmt(ex)}</span></div>
    <div class="pl-row"><span class="pl-label">🛒 Purchases</span><span class="pl-value tr">₹${fmt(pur)}</span></div>
    <div class="pl-row"><span class="pl-label">⚰️ Deaths</span><span class="pl-value tr">₹${fmt(dl)}</span></div>
    <div class="pl-row"><span class="pl-label fw7" style="color:${net>=0?'var(--green)':'var(--red)'}">NET ${net>=0?'PROFIT':'LOSS'}</span><span class="pl-value" style="font-size:20px;color:${net>=0?'var(--green)':'var(--red)'}">₹${fmt(Math.abs(net))}</span></div>
  </div>`;
}

function buildOutstanding(m){
  const mk=(DB.milkCust||[]).filter(r=>r.month===m);
  const sl=(DB.slaughter||[]).filter(r=>r.month===m);
  const cb=(DB.careBills||[]).filter(r=>r.month===m);
  const tr=(DB.transport||[]).filter(r=>r.month===m);
  const lb=(DB.salaries||[]).filter(r=>r.month===m);
  const pur=(DB.purchases||[]).filter(r=>r.month===m);
  const mkDue=mk.reduce((s,r)=>s+r.value-r.received,0);
  const slDue=sl.reduce((s,r)=>s+r.amount-r.received,0);
  const cbDue=cb.reduce((s,r)=>s+r.bill-r.paid,0);
  const trDue=tr.reduce((s,r)=>s+r.total-r.paid,0);
  const lbDue=lb.reduce((s,r)=>s+r.sal-(r.adv||0)-(r.bal||0),0);
  const purDue=pur.reduce((s,r)=>s+r.amount-r.paid,0);
  return `<div class="pl-block">
    <div class="pl-row"><span class="pl-label">🥛 Milk Receivable</span><span class="pl-value ${mkDue>0?'tr':'tg'}">₹${fmt(mkDue)}</span></div>
    <div class="pl-row"><span class="pl-label">🔪 Slaughter Receivable</span><span class="pl-value ${slDue>0?'tr':'tg'}">₹${fmt(slDue)}</span></div>
    <div class="pl-row"><span class="pl-label">👤 Caretaker Due</span><span class="pl-value ${cbDue>0?'tr':'tg'}">₹${fmt(cbDue)}</span></div>
    <div class="pl-row"><span class="pl-label">🚛 Transport Due</span><span class="pl-value ${trDue>0?'tr':'tg'}">₹${fmt(trDue)}</span></div>
    <div class="pl-row"><span class="pl-label">👷 Labour Due</span><span class="pl-value ${lbDue>0?'tr':'tg'}">₹${fmt(lbDue)}</span></div>
    <div class="pl-row"><span class="pl-label">🛒 Purchase Due</span><span class="pl-value ${purDue>0?'tr':'tg'}">₹${fmt(purDue)}</span></div>
  </div>`;
}

function buildMilkBreakdown(m){
  const recs=(DB.milkCust||[]).filter(r=>r.month===m);
  const byC={};
  recs.forEach(r=>{ if(!byC[r.customer])byC[r.customer]={l:0,v:0,r:0}; byC[r.customer].l+=r.litres;byC[r.customer].v+=r.value;byC[r.customer].r+=r.received; });
  if(Object.keys(byC).length===0) return '<div class="card"><div class="empty"><div class="et">No milk records</div></div></div>';
  return `<div class="card"><table class="tbl">
    <thead><tr><th>Customer</th><th>Route</th><th>Litres</th><th>Value</th><th>Received</th><th>Due</th></tr></thead>
    <tbody>${Object.entries(byC).map(([c,v])=>`<tr>
      <td class="fw7">${c}</td>
      <td><span class="badge ${C.RATES[c]===2.20?'b-amber':'b-green'}">${C.RATES[c]===2.20?'A':'B'}</span></td>
      <td class="mono">${v.l}L</td>
      <td class="mono tg">₹${fmt(v.v)}</td>
      <td class="mono tm">₹${fmt(v.r)}</td>
      <td class="mono ${v.v-v.r>0?'tr':'tg'}">₹${fmt(v.v-v.r)}</td>
    </tr>`).join('')}</tbody>
    <tfoot><tr><td colspan="2">TOTAL</td><td class="mono">${recs.reduce((s,r)=>s+r.litres,0)}L</td><td class="mono tg">₹${fmt(recs.reduce((s,r)=>s+r.value,0))}</td><td class="mono tm">₹${fmt(recs.reduce((s,r)=>s+r.received,0))}</td><td class="mono tr">₹${fmt(recs.reduce((s,r)=>s+r.value-r.received,0))}</td></tr></tfoot>
  </table></div>`;
}

/* ═══════════════════════════════════════════════════════
   YEARLY REPORT
═══════════════════════════════════════════════════════ */
function renderYearly(el){
  let totInc=0,totExp=0,totLoss=0;
  const rows=C.MONTHS.map(m=>{
    const mk=(DB.milkCust||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.value,0);
    const sl=(DB.slaughter||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.amount,0);
    const ex=(DB.expenses||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.amount,0);
    const cb=(DB.careBills||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.bill,0);
    const tr=(DB.transport||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.total,0);
    const lb=(DB.salaries||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.sal,0);
    const pur=(DB.purchases||[]).filter(r=>r.month===m).reduce((s,r)=>s+r.amount,0);
    const dl=(DB.deaths||[]).filter(r=>r.date.startsWith(m)).reduce((s,r)=>s+r.val,0);
    const inc=mk+sl, exp=ex+cb+tr+lb+pur, net=inc-exp-dl;
    totInc+=inc; totExp+=exp; totLoss+=dl;
    return {m,inc,exp,dl,net};
  });
  const yearNet=totInc-totExp-totLoss;

  el.innerHTML=`
    <div class="sec-label">📅 Year 2026 — Annual P&L</div>
    <div class="pl-block" style="margin-bottom:20px">
      <div class="pl-row"><span class="pl-label fw7 tg">TOTAL INCOME</span><span class="pl-value tg" style="font-size:18px">₹${fmt(totInc)}</span></div>
      <div class="pl-row"><span class="pl-label fw7 tr">TOTAL EXPENSE</span><span class="pl-value tr" style="font-size:18px">₹${fmt(totExp)}</span></div>
      <div class="pl-row"><span class="pl-label fw7 tr">DEATH LOSSES</span><span class="pl-value tr">₹${fmt(totLoss)}</span></div>
      <div class="pl-row"><span class="pl-label fw7" style="font-size:15px;color:${yearNet>=0?'var(--green)':'var(--red)'}">NET YEAR ${yearNet>=0?'PROFIT':'LOSS'}</span><span class="pl-value" style="font-size:24px;color:${yearNet>=0?'var(--green)':'var(--red)'}">₹${fmt(Math.abs(yearNet))}</span></div>
    </div>

    <div class="sec-label">Month-by-Month Breakdown</div>
    <div class="card">
      <table class="tbl">
        <thead><tr><th>Month</th><th>Income</th><th>Expense</th><th>Deaths</th><th>Net</th><th>Status</th></tr></thead>
        <tbody>${rows.map(r=>`<tr class="${r.net<0?'r-alert':'r-ok'}">
          <td class="fw7">${mll(r.m)}</td>
          <td class="mono tg">₹${fmt(r.inc)}</td>
          <td class="mono tr">₹${fmt(r.exp)}</td>
          <td class="mono tr">₹${fmt(r.dl)}</td>
          <td class="mono fw7 ${r.net>=0?'tg':'tr'}">₹${fmt(Math.abs(r.net))}</td>
          <td>${r.net>=0?'<span class="badge b-green">PROFIT</span>':r.inc===0?'<span class="badge b-gray">NO DATA</span>':'<span class="badge b-red">LOSS</span>'}</td>
        </tr>`).join('')}</tbody>
        <tfoot><tr><td class="fw7">YEAR TOTAL</td><td class="mono tg">₹${fmt(totInc)}</td><td class="mono tr">₹${fmt(totExp)}</td><td class="mono tr">₹${fmt(totLoss)}</td><td class="mono fw7 ${yearNet>=0?'tg':'tr'}">₹${fmt(Math.abs(yearNet))}</td><td>${yearNet>=0?'<span class="badge b-green">PROFIT</span>':'<span class="badge b-red">LOSS</span>'}</td></tr></tfoot>
      </table>
    </div>`;
}
